  12 // Phone recordings use M4A; browsers may produce WebM or Ogg even when a
  13 // client labels them differently. Sniff the bytes to send the true format.
  14 function detectAudioFormat(bytes: Uint8Array, fallbackMime: string): string {
  15   if (bytes.length >= 4) {
  16     // "OggS"
  17     if (bytes[0] === 0x4f && bytes[1] === 0x67 && bytes[2] === 0x67 && bytes[3] === 0x53) {
  18       return 'ogg';
  19     }
  20     // EBML header (WebM/Matroska)
  21     if (bytes[0] === 0x1a && bytes[1] === 0x45 && bytes[2] === 0xdf && bytes[3] === 0xa3) {
  22       return 'webm';
  23     }
  24     // "RIFF" (WAV)
  25     if (bytes[0] === 0x52 && bytes[1] === 0x49 && bytes[2] === 0x46 && bytes[3] === 0x46) {
  26       return 'wav';
  27     }
  28     // "fLaC"
  29     if (bytes[0] === 0x66 && bytes[1] === 0x4c && bytes[2] === 0x61 && bytes[3] === 0x43) {
  30       return 'flac';
  31     }
  32     // ID3 tag or MPEG frame sync (MP3)
  33     if (
  34       (bytes[0] === 0x49 && bytes[1] === 0x44 && bytes[2] === 0x33) ||
  35       (bytes[0] === 0xff && (bytes[1] & 0xe0) === 0xe0)
  36     ) {
  37       return 'mp3';
  38     }
  39     // "ftyp" at offset 4 (MP4/M4A)
  40     if (
  41       bytes.length >= 8 &&
  42       bytes[4] === 0x66 && bytes[5] === 0x74 && bytes[6] === 0x79 && bytes[7] === 0x70
  43     ) {
  44       return 'mp4';
  45     }
  46   }
  47 
  48   const mime = fallbackMime.toLowerCase();
  49   if (mime.includes('ogg')) return 'ogg';
  50   if (mime.includes('wav')) return 'wav';
  51   if (mime.includes('mp3') || mime.includes('mpeg')) return 'mp3';
  52   if (mime.includes('mp4') || mime.includes('m4a')) return 'mp4';
  53   if (mime.includes('flac')) return 'flac';
  54   return 'webm';
  55 }
